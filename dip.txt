    ################################################## Low Pass Filter ##################################################import numpy as np

    import matplotlib.pyplot as plt
    import cv2

    def lpf(img , filters, d0, n):
        f_trans = np.fft.fft2(img)
        f_shift = np.fft.fftshift(f_trans)

        rows,cols = img.shape
        u,v = np.meshgrid(np.arange(cols)-cols//2,np.arange(rows)-rows//2)
        d = np.sqrt(u**2+v**2)

        if filters == 'ideal':
            mask = np.where(d<=d0,1,0)
        elif filters == "butterworth":
            mask = 1/(1+(d/d0)**2*n)
        elif filters == "gaussian":
            mask = np.exp(-(d**2)/(2*(d0**2)))
        else:
            raise ValueError("Invalid filter type. Choose 'ideal', 'butterworth', or 'gaussian'")

        f_filtered = f_shift * mask
        f_ishift = np.fft.ifftshift(f_filtered)
        f_itrans = np.fft.ifft2(f_ishift)
        img_back = np.abs(f_itrans)

        return img_back
    img = cv2.imread("lion.jpg",0)
    ideal = lpf(img , "ideal", 50, 2)
    butterworth = lpf(img , "butterworth", 50, 2)
    gaussian = lpf(img , "gaussian", 50, 2)

    result = [img,ideal,butterworth,gaussian]
    titles = ["Original","Ideal","Butter Worth","Gaussian"]

    plt.figure(figsize=(20,4))
    for i in range(4):
        plt.subplot(1,4,i+1)
        plt.imshow(result[i],cmap="gray")
        plt.title(titles[i])
        plt.axis("off")
    plt.show()

    import numpy as np
    import matplotlib.pyplot as plt

    def hpf(img , filters, d0, n):

        rows,cols = img.shape
        u,v = np.meshgrid(np.arange(cols)-cols//2,np.arange(rows)-rows//2)
        d = np.sqrt(u**2+v**2)

        if filters == 'ideal':
            mask = np.where(d>d0,1,0)
        elif filters == "butterworth":
            mask = 1/(1+(d0/(d+1e-5))**2*n)
        elif filters == "gaussian":
            mask = 1- np.exp(-(d**2)/(2*(d0**2)))
        else:

    import numpy as np
    import matplotlib.pyplot as plt
    import cv2

    def Region_Growing(img,seed,threshold):
        rows, cols = img.shape

        segmented = np.zeros((rows,cols),dtype = np.uint8)
        visited = np.zeros((rows,cols),dtype = bool)
        stack = [seed]
        seed_values = int(img[seed]) 
        segmented[seed] = 255 
        visited[seed] = True

        while stack:
            x,y = stack.pop()
            segmented[x,y]=255
            for nx,ny in [(x-1,y),(x+1,y),(x,y-1),(x,y+1)]:
                if 0<=nx<rows and 0<=ny<cols and not visited[nx,ny]:
                    if np.abs(int(img[nx,ny])-seed_values) <= threshold:
                        visited[nx,ny] = True
                        stack.append((nx,ny))
        return segmented


    img = cv2.imread("Lenna.png",0)
    seed = (100,100)
    threshold = 30
    result = Region_Growing(img,seed,threshold)
    kernel = np.ones((5,5),np.uint8)
    result = cv2.morphologyEx(result,cv2.MORPH_CLOSE,kernel)

    plt.figure(figsize=(30,15))
    plt.subplot(1,4,1)
    plt.imshow(img,cmap = "gray")

    import numpy as np
    import matplotlib.pyplot as plt
    import cv2

    def watershed(image_path):
        image_nongray = cv2.imread(image_path)
        image = cv2.cvtColor(image_nongray,cv2.COLOR_BGR2GRAY)

        _,thresh = cv2.threshold(image,0,255,cv2.THRESH_BINARY_INV+cv2.THRESH_OTSU)
        kernel = np.ones((3,3),np.uint8)
        opening = cv2.morphologyEx(thresh,cv2.MORPH_OPEN,kernel,iterations=2)
        
        sure_bg = cv2.dilate(opening,kernel,iterations=3)
        dist_transform = cv2.distanceTransform(opening,cv2.DIST_L2, 5)
        _,sure_fg = cv2.threshold(dist_transform,0.7*dist_transform.max(),255,cv2.THRESH_BINARY)
        sure_fg= np.uint8(sure_fg)
        
        unknown = np.subtract(sure_bg,sure_fg)
        _,markers = cv2.connectedComponents(sure_fg)
        markers = markers+1
        markers[unknown==255] = 0

        markers = cv2.watershed(image_nongray, markers)
        image_nongray[markers==-1] = [255,0,0]

        plt.figure(figsize=(8, 6))
        plt.imshow(cv2.cvtColor(image_nongray, cv2.COLOR_BGR2RGB))
        plt.title("Watershed Segmentation")
        plt.axis('off')
        plt.show()
        
    if __name__ == "__main__":
        watershed("Lenna.png")
